  # 1. PAM development headers, so the build can compile the small C login helper
  sudo apt install libpam0g-dev

  # 2. Build everything (this also builds packages/gateway/dist/pam-helper)
  cd ~/nebula && yarn workspaces foreach -A --parallel --topological-dev run build

  # 3. Copy the build somewhere every user can read (your home folder is private: 0750)
  sudo rsync -a --delete ~/nebula/ /opt/nebula/

  # 4. A node binary every user can run (yours comes from nvm, inside your private home)
  sudo cp "$(command -v node)" /usr/local/bin/node

  # 5. Register "nebula" as a PAM login service (same as sshd has /etc/pam.d/sshd)
  sudo cp /opt/nebula/packages/gateway/pam/nebula /etc/pam.d/nebula

  # 6. Unprivileged account the web-facing process runs as (same idea as sshd's "sshd" user)
  sudo useradd --system --no-create-home --shell /usr/sbin/nologin nebula

  # 7. A second account to check that users are isolated (optional but worth it)
  sudo useradd -m -s /bin/bash testuser && sudo passwd testuser

  # 8. Start the gateway as root (the root part only does login and session startup)
  sudo /usr/local/bin/node /opt/nebula/packages/gateway/dist/main.js \
    --bind-port 8443 --applications=$HOME/nebula-apps.json
